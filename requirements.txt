# =============================================================================
# WHAM Pipeline Requirements
# =============================================================================
# Full pipeline: video preprocessing → WHAM inference → beta/theta extraction
#
# INSTALL ORDER
# ─────────────────────────────────────────────────────────────────────────────
# 1. Create and activate the conda environment:
#      conda create -n wham_dev python=3.9 -y
#      conda activate wham_dev
#
# 2. Install PyTorch with CUDA 12.8 (matches RTX 5090 / sm_120):
#      pip install torch==2.7.0+cu128 torchvision==0.22.0+cu128 torchaudio==2.7.0+cu128 \
#          --index-url https://download.pytorch.org/whl/cu128
#
# 3. Install torch-scatter (must match torch + CUDA version exactly):
#      pip install torch-scatter==2.1.2+pt27cu128 \
#          -f https://data.pyg.org/whl/torch-2.7.0+cu128.html
#
# 4. Install pytorch3d (no pre-built wheel for PyTorch 2.7 — build from source):
#      TORCH_CUDA_ARCH_LIST="8.0;8.6;8.9;9.0+PTX" \
#      FORCE_CUDA=1 \
#      pip install "git+https://github.com/facebookresearch/pytorch3d.git" \
#          --no-build-isolation
#
# 5. Install DPVO (local build — patches required, see README):
#      TORCH_CUDA_ARCH_LIST="8.0;8.6;8.9;9.0+PTX" \
#      pip install third-party/DPVO --no-build-isolation
#
# 6. Install everything else:
#      pip install -r requirements.txt
#
# NOTES
# ─────────────────────────────────────────────────────────────────────────────
# - torch, torchvision, torchaudio, torch-scatter, pytorch3d, and dpvo are
#   NOT listed below because they require special install commands (above).
# - mmcv-full 1.7.2 is used instead of the requirements.txt mmcv==1.3.9;
#   the ViTPose version check in third-party/ViTPose/mmpose/__init__.py is
#   patched to accept up to 1.7.2.
# - setuptools is pinned to 59.5.0 — newer versions break mmcv-full builds.
# - All torch.load() calls use weights_only=False (PyTorch 2.6+ compat patch).
# =============================================================================
